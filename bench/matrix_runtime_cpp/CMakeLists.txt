cmake_minimum_required(VERSION 3.18)
project(cgx2_matrix_baseline LANGUAGES CXX)
find_package(Torch REQUIRED)
if(NOT CGX2_EXPECT_TORCH_VERSION STREQUAL "2.11.0")
  message(FATAL_ERROR "Set CGX2_EXPECT_TORCH_VERSION=2.11.0")
endif()
add_executable(cgx2_matrix_baseline main.cpp)
target_compile_features(cgx2_matrix_baseline PRIVATE cxx_std_20)
target_compile_options(cgx2_matrix_baseline PRIVATE ${TORCH_CXX_FLAGS})
target_link_libraries(cgx2_matrix_baseline PRIVATE ${TORCH_LIBRARIES} dl)
